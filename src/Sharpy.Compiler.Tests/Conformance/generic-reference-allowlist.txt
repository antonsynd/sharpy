# ── #1192 — generic TYPE references are neither arity-checked nor rejected uncalled ──
# `_x = Outer.Inner[int]` emits `Outer.Inner[global::Sharpy.Builtins.Int]` (CS0021/CS0119) and
# `Outer.Inner[int, str](5)` emits `new Outer.Inner<int, string>(5)` (CS0305). Reproduces identically
# for a bare `Box[int]`, so the fix is class-level across GenericTypeRef / ModuleType / NestedTypeRef
# and must respect PEP-696 type-parameter defaults.

nested_outer_inner::uncalledAssigned::excess
nested_outer_inner::uncalledThenCalled::exact
nested_outer_inner::passedAsArg::exact
nested_outer_inner::ternaryCalled::exact
nested_outer_inner::called::excess
nested_outer_pair::uncalledAssigned::exact

# ── #1193 — a nested generic type constructed without explicit type args gets no inference ──
# `Outer.Inner(5)` emits `new Outer.Inner(5)` (CS0305) where the bare spelling `Box(5)` infers
# `new Box<int>(5)`: the construction path's member-access arm resolves only MODULE qualifiers.
nested_outer_inner::called::none
nested_outer_inner::parenCalled::none
nested_outer_pair::called::none
nested_outer_pair::parenCalled::none

# Unknown, so nothing proves absence — though for a user-declared type the symbol proves it outright.
nested_outer_inner::typoMember::exact
nested_outer_pair::typoMember::exact
